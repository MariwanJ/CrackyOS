186/* This function will remove the extra '\' from the following commands that found in the script program sent to the system.
187 Without that, they will be shown as they are and the commands they represent will not be executed by printf.
188* \' single quote
189* \" double quote
190* \? question mark
191* \\ backslash
192* \a audible bell
193* \b backspace
194* \f form feed -
195* \n line feed -
196* \r carriage return
197* \t horizontal tab
198* \v vertical tab
199*@param InputString : message should be fixed
200*/
201char* correctString(const char* InputString)
202{
203 //Stop running if we have error
204 STOP_CODE_EXECUTION_NULL
205
206 unsigned int i = 0;
207 unsigned int length = strlen(InputString);
208 char* result = (char*)calloc(length + 1, sizeof(char));
209 unsigned int offset = 0;
210 while ((i + offset) <= length) {
211 if ((offset + i + 1) <= length) {
212 if ((InputString[i] == '\\') && (InputString[i + 1] == 't')) {
213 result[i] = '\t';
214 offset++;
215 }
216 else if ((InputString[i] == '\\') && (InputString[i + 1] == 'n')) {

223 }

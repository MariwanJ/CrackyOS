84int SERIAL_Print(const char *format, ...) {
85 int len = 0;
86 char *pstrBuf = NULL;
87 char *scanBuffr = NULL;
88 std::va_list arg;
89 va_start(arg, format);
90 std::va_list length_arg;
91 va_copy(length_arg, arg);
92 // ARMCC microlib does not properly handle a size of 0.
93 // As a workaround supply a dummy buffer with a size of 1.
94 char dummy_buf[1];
95 len = vsnprintf(dummy_buf, sizeof(dummy_buf), format, length_arg);
96 va_end(length_arg);
97 pstrBuf = new char[len + 1];
98 if (pstrBuf == NULL) {
99 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_MEM_ALLOCATION_ERROR);
100 return quitScriptProgram(1);
101 }
102 vsnprintf(pstrBuf, len + 1, format, arg);
103 scanBuffr = pstrBuf;
104 while (*scanBuffr != '\0') {
105 globalSerialCOMvar.Serial_pc->write(scanBuffr, 1);
106 scanBuffr++;
107 }
108 // printf(pstrBuf); We avoid blocking the execution. A thread will take care of this part.
109 va_end(arg);
110 delete[] pstrBuf;
111 pstrBuf = NULL;
112 return len;
113}

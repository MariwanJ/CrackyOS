180inline float extractFloat(char *stringToFloat) {
181 // Stop running if we have error
182 float Result = 0;
183 float fact = 1;
184 if (*stringToFloat == '-') {
185 stringToFloat++;
186 fact = -1;
187 };
188 for (int pointFound = 0; *stringToFloat; stringToFloat++) {
189 if (*stringToFloat == '.') {
190 pointFound = 1;
191 continue;
192 };
193 int d = *stringToFloat - '0';
194 if (d >= 0 && d <= 9) {
195 if (pointFound)
196 fact *= 0.1000000000f;
197 Result = Result * 10.0000000000f + (float)d;
198 };
199 };
200 return (Result * fact);
201}

262inline uint32_t CalculateHashOfString(const char* pText, unsigned int length)
263{
264 if (length > MAX_HASH_STRING_LENGTH) {
265 StopExecution = 1; // MAX number of bytes is 16 bytes.
266 return 0;
267 }
268
269 if (pText == NULL) {
270 return 0; // ERROR
271 }
272
273 uint32_t h = 2166136261UL; // FNV-1a offset basis
274
275 for (unsigned int i = 0; i < length; i++) {
276
277 if (pText[i] == '\0') {
278 return 0; // ERROR
279 }
280
281 h ^= (uint8_t)pText[i];
282 h *= 16777619UL; // FNV-1a prime
283 }
284
285 return h;
286}

305inline float convert_text_to_number(void) {
306 // Stop running if we have error
307 float result;
308
309 /*Possible combination of text and numbers in a string (it should be at least
310 3 char and max 10) 0x00 ---> 0xFF 0B00000000 ---> 0B11111111
311 0O00 ---> 0O56
312 000 ---> 255
313 any thing else will generate an error!
314 */
315 switch (*(p_prog->txtString + 1)) {
316 case 'x': {
317 // Hex number
318 result = (float)strtol(p_prog->txtString, NULL,
319 16); // Convert a hex number to long
320 break;
321 }
322 case 'b': {
323 // Binary number
324 result = (float)strtol((p_prog->txtString + 2), NULL,
325 2); // Convert a binary number to long.. strtol
326 // cannot take 0b .. remove it from the buffer
327 break;
328 }
329 case 'o': {
330 // octal number
331 result = (float)strtol((p_prog->txtString + 2), NULL,
332 8); // Convert a octal number to long.. strtol cannot
333 // take 0o .. remove it from the buffer
334 break;
335 }
336
337 default: {
338 result =
339 extractFloat(p_prog->txtString); // Convert the first digits found in
340 // the string to decimal numbers.
341 break;
342 }
343 }
344 return result;
345}

347inline int check_and_get_next_token(void) {
348 /* New way to do this job is like this:
349 1-get the new position from p_prog_next
350 2-give the next position to p_prog_next
351 3-Check if it is TOKEN_NEW_LINE ..
352 4-Check if the new token is enum of the TOKEN_WORD
353 5-If so - take the number and save it in current_token_word
354 6-give the get the new position from p_prog_next and put it in p_prog
355 7-give the next position to p_prog_next
356 */
357
358 p_prog = p_prog_next; // If it is not TOKEN_NEW_LINE, this will bring only the
359 // enum part of the token.
360 if (end_of_program_token() == 1) {
361 current_token_word = 1; // TOKEN_EOF=1
362 return current_token_word; // We return 0 whenever we are done EOF
363 }
364 // New .dad file shouldn't have any space, neither new line at the end of the
365 // instructions or arguments.
366 if (*p_prog->txtString == '\n') {
367 // \n causes to have two '\n\n .. since the text is equal to '\n' Skip it so
368 // we don't have a problem later.
369 p_prog = p_prog->next;
370 }
371 p_prog_next = p_prog->next;
372 // Now we have to check if the token is digits (numbers) : Dec, Hex, bin are
373 // supported.
374 if (isdigit(*p_prog->txtString)) {
375 // Convert the token-text to int
376 current_token_word = (int)convert_text_to_number();
377
378 p_prog = p_prog_next;
379 p_prog_next =
380 p_prog->next; // Since we have always 2 records per each token ..
381 // we need to move the next cursor to the new position one more time.
382 return current_token_word;
383 } else {
384 // We shouldn't be here .. Each time when a token saves in the dad file ..
385 // first is the token_enum then it is the real text
386 // DEBUG_MESSAGE("ERROR %i\n",DBG_MSG_DAD_FILE_CORRUPTED);
387 return quitScriptProgram(1);
388 }
389 return 0;
390}

143void TickerObject_add(int enumPinName, uint32_t uSecTime, bool DefaultValue) {
144 tobj* myObj, * oldObj;
145 myObj = GTickerObject;
146
147 oldObj = TickerObject_find(enumPinName);
148 if (oldObj == NULL) {
149 if (myObj == NULL) {
150 //No object defined .. create the first
151 myObj = new tobj;
152 if (myObj == NULL) {
153 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_MEM_ALLOCATION_ERROR_TICKER);
154 quitScriptProgram(1);
155 return;
156 }
157 //Reset the pointers.
158 myObj->next = myObj->prev = NULL;
159 GTickerObject = myObj;
160 }
161 else
162 {
163 while (myObj->next != NULL) myObj = myObj->next;
164 myObj->next = new tobj;
165 if (myObj->next == NULL) {
166 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_MEM_ALLOCATION_ERROR_TICKER);
167 quitScriptProgram(1);
168 return;
169 }
170 myObj->next->prev = myObj;
171 myObj->next->next = NULL;
172 myObj = myObj->next;
173 }
174 //Create new digital out pin.
175 myObj->mToggelPin = new Flipper(enumPinName);
176 if (myObj->mToggelPin == NULL) {
177 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_MEM_ALLOCATION_ERROR_TICKER_DIGITALOUT);
178 quitScriptProgram(1);
179 return;
180 }
181
182 myObj->mToggelPin->m_pin = addGPIOpin_DIGITAL(myObj->mToggelPin->enumPINNAME(), GPIO_TICKER);
183 if (myObj->mToggelPin->m_pin == NULL) {
184 //We have a bad instruction.
185 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_SYNTAX_ERROR_TICKER);
186 quitScriptProgram(1);
187 return;
188 }
189 }
190 else {
191 myObj = oldObj; //We found the old object.
192 }
193 myObj->mToggelPin->setInterval(uSecTime);
194 myObj->mToggelPin->__detach();// We do this always as we might need to update the interval
195 if (DefaultValue) {
196 myObj->mToggelPin->__attach();
197 }
198}

205int token_ticker() {
206 float interval = 0.0;
207 bool start = 1;
208 process_token(TOKEN_TICKER);
209 if (current_token_word == TOKEN_COMMA)
210 process_token(TOKEN_COMMA);
211 //1) We should take the pin name here.
212
213 int tempenumPin = current_token_word;
214 PinName tempPin = retrievePinNameFromTOKEN_NAME(current_token_word);
215 if (tempPin == NC) {
216 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_TICKER_PINNAME);
217 return quitScriptProgram(1);
218 }
219
220 process_token(current_token_word);
221 if (current_token_word == TOKEN_COMMA)
222 process_token(TOKEN_COMMA);
223
224 //2) Interval
225 if (current_token_word == TOKEN_NUMBER) {
226 interval = convert_text_to_number();
227 process_token(TOKEN_NUMBER);
228 }
229 else if (current_token_word == TOKEN_VARIABLE) {
230 int varPosition = TestSystem_get_variable(NULL); // we don't need to send the name .. the function will do the job.
231 process_token(TOKEN_VARIABLE);
232 if (varPosition < 0) {
233 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_SYNTAX_ERROR_TICKER);
234 quitScriptProgram(1); //Exit the program.
235 }
236 if (script_variables[varPosition].tag != T_INT) {
237 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_SYNTAX_ERROR_TICKER);
238 quitScriptProgram(1);
239 }
240 interval = *script_variables[varPosition].Value.pi;
241 }
242 else {
243 //Syntax error. .only number is allowed
244 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_TICKER_INTERVAL);
245 return quitScriptProgram(1);
246 }
247
248 if (current_token_word == TOKEN_COMMA)
249 process_token(TOKEN_COMMA);
250 //3)Start/Stop
251 //This is optional. Default is the ticker will start immediately
252 if (current_token_word == TOKEN_NEW_LINE ||
253 current_token_word == TOKEN_SEMICOLON ||
254 current_token_word == TOKEN_EOF) {
255 start = 1; //We are done.
256 }
257 else
258 {
259 //4) Start or Stop
260
261 if (current_token_word != TOKEN_NUMBER) {
262 //Syntax error
263 DEBUG_MESSAGE("ERROR %i\n", DBG_MSG_TICKER_START_STOP);
264 return quitScriptProgram(1);
265 }
266 start = convert_text_to_number();
267 process_token(TOKEN_NUMBER);
268 }
269 //We have everything. Create the object and register the GPIO.
270 TickerObject_add(tempenumPin, interval, start);
271 return 0;
272}
